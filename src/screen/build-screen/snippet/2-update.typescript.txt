const update = (msg: Msg, model: Model): [Model, Cmd<Msg>] => {
  switch (msg._tag) {
    case 'Increment':
    case 'Tick':
      return [{ ...model, count: model.count + 1 }, Cmd.none()]
    case 'FetchQuote':
      // A description of the request: the runtime runs it, not update
      return [
        model,
        Cmd.attempt(fetchQuote, (result): Msg => ({ _tag: 'QuoteFetched', result })),
      ]
    case 'QuoteFetched':
      return [
        msg.result.ok ? { ...model, quote: msg.result.value } : model,
        Cmd.none(),
      ]
  }
}
